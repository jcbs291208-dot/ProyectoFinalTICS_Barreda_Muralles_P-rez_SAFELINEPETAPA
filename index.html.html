Código C++ (Arduino)


// ==========================================
// PROYECTO FINAL: IRTRA PETAPA
// Autores: [Apellidos del Equipo][span_9](start_span)[span_9](end_span)
// ==========================================

const int pinSensor = A0; 
const int pinLed = 13;

void setup() {
  pinMode(pinLed, OUTPUT);
  Serial.begin(9600);
  // Inicialización de componentes
}

void loop() {
  int lectura = analogRead(pinSensor);
  
  if (lectura > 500) {
    digitalWrite(pinLed, HIGH); // Activar sistema de alerta
  } else {
    digitalWrite(pinLed, LOW);
  }
  
  delay(100);
}
